// Regole Firestore suggerite per le collection usate dai giochi FaW.
//
// CONTESTO VERIFICATO (leggere prima di incollarle):
//   • Nel repository NON esiste alcun file di regole Firestore. L'unico file di regole
//     presente è games/Pong/database.rules.json (Realtime Database). Cosa sia online sul
//     progetto `funatwork-cd237` non è ispezionabile da qui: va letto nella console.
//   • Firebase Authentication È ATTIVA nel progetto (`authDomain` valorizzato in
//     games/shared/firebase-config.js), ma nessuna pagina del sito chiama l'SDK Auth:
//     grep su index.html, games/*/js e games/Pong → zero occorrenze di firebase.auth(),
//     signIn*, currentUser, getIdToken. L'accesso del portale è l'hardcoded
//     `utenti/{NOME}` con hash SHA-256 client-side.
//     Conseguenza: `request.auth` è null per TUTTI i client di questo sito, quindi una
//     regola `request.auth != null` bloccherebbe immediatamente Ruzzle, il Gioco del 15,
//     l'hub e i tre giochi nuovi. È per questo che le regole odierne devono essere lasse.
//   • Da questo lavoro esiste un opt-in spento di default: FAWNet.ensureSignedIn()
//     (`?auth=anon` o localStorage faw:auth:anon="1") fa il login anonimo, espone
//     FAWNet.uid() e FAWRoom.buildMatch scrive `authUid` nel documento partita.
//   • App Check: l'attivazione in index.html (riga ~673) è COMMENTATA. Se la console ha
//     l'enforcement di App Check attivo su Firestore, ogni scrittura del sito viene
//     rifiutata — verifica lì prima di toccare le regole.
//
// Come provarle SENZA toccare la produzione:
//   firebase emulators:start --only firestore
//   firebase deploy --only firestore:rules --project <progetto-di-prova>
// I test Playwright di questo repo girano contro tests/support/faw-relay.js (mini-Firestore
// in memoria), quindi non esercitano le regole: una suite di regole va scritta a parte
// (firestore-tests con @firebase/rules-unit-testing, che qui non è installato).
//
// CAMPI REALI di `partite/{id}` (letti in games/shared/faw-room.js e nei tre giochi):
//   gioco, stato ('attesa'|'pronto'|'in_corso'|'chiusura'|'risultati'|'conclusa'|'annullata'),
//   host, creator, partecipanti[String], giocatori{}, pronti[String], punteggi{}, parole{},
//   opzioni{durata,mode,countdown,griglia,miccia}, seed, cfg{griglia,bomba},
//   durata, createdAt, timestamp, dataOra, authUid, startAt, endsAt, fine, claim{},
//   log[], archive, risultati{}, rivincitaDi, rivincitaAccettataDa[], rivincitaRifiutataDa[],
//   round, + arena{} / rush{} / bomba{}
//   NB: `partecipanti`, `host`, `creator`, `claim.chi` e le chiavi di `punteggi` sono NOMI,
//   non uid. Le regole non possono quindi "verificare che sei tu": possono solo validare la forma, finché non si firma in anonimo.

service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- helper ----------
    function statoGiusto() {
      return !('stato' in request.resource.data) || request.resource.data.stato in
        ['attesa', 'pronto', 'in_corso', 'chiusura', 'risultati', 'conclusa', 'annullata'];
    }

    function durataSana() {
      // i giochi stanno tra 120 000 e 240 000 ms; il campo è in millisecondi
      return !('durata' in request.resource.data) ||
             (request.resource.data.durata is number &&
              request.resource.data.durata >= 30000 &&
              request.resource.data.durata <= 1800000);
    }

    function documentoPiccolo() {
      // le liste di gioco sono capate dal codice (log 30, parole 200, usate 400):
      // qui si blocca solo la deriva, non si fa sicurezza
      return request.resource.data.keys().size() <= 40;
    }

    function nonCambiaIdentita() {
      // I campi decisi alla creazione restano fermi: identità della partita, regole
      // scelte, catena di rivincite. NOTA: `cfg.griglia` è scritto una volta all'avvio
      // da Parole in Arena (griglia generata dal seed), quindi `cfg` non può stare tra
      // i campi fissi; `seed` è un campo proprio di buildMatch e resta fisso.
      let fissi = ['gioco', 'createdAt', 'timestamp', 'seed', 'opzioni', 'rivincitaDi', 'authUid'];
      return fissi.all(k => !(k in request.resource.data) ||
                             !(k in resource.data) ||
                             resource.data[k] == request.resource.data[k]);
    }

    // ---------- VARIANTE A — compatibile con l'assetto attuale ----------
    // Cosa garantisce DAVVERO: solo validazione strutturale. Un client bugdato non può
    // più corrompere una partita (stato sconosciuto, durata assurda, documento gigante,
    // regole riscritte a metà). Un client che vuole barare può continuare a farlo:
    // senza login non esiste un'identità da controllare.
    // Se le regole online sono più lasse di così (tipicamente `allow read, write: if true`)
    // questa variante è un miglioramento senza rompere nulla.
    match /partite/{partitaId} {
      allow read: if true;                       // le partite sono raggiungibili per codice-invito
      allow create: if statoGiusto() && durataSana() && documentoPiccolo();
      allow update: if statoGiusto() && durataSana() && documentoPiccolo() && nonCambiaIdentita();
      // L'hub offre il pulsante «🗑️ elimina» in 5 punti (index.html:1271, 1277, 1303,
      // 1335, 1364) e cancella `partite/{id}`: negare il delete romperebbe l'hub.
      allow delete: if true;
    }

    // Presenza: heartbeat breve. Oggi è scrivibile da chiunque sappia il nome utente:
    // è un limite noto, non un obiettivo raggiunto.
    match /presenze/{userId} {
      allow read: if true;
      allow write: if request.resource.data.keys().size() <= 10;
    }

    // Amicizie/inviti sociali dell'hub: `amicizie/{A}_{B}` con liste di nomi.
    match /amicizie/{coppia} {
      allow read: if true;
      allow write: if request.resource.data.keys().size() <= 12;
    }

    // Utenti (account del portale + statistiche). Non toccato da questo lavoro.
    // `passwordHash` è un hash SHA-256 di una password scelta dall'utente, senza salt:
    // con la lettura pubblica è offline-crackable. È il motivo per cui la variante B
    // (o una regola `allow get: if false`) andrebbe applicata PRESTO.
    match /utenti/{userId} {
      allow read: if true;
      allow create, update: if request.resource.data.keys().size() <= 60;
      allow delete: if false;
    }

    // `pictionary_rooms` e le chiavi RTDB di Pong restano fuori scope di questa bozza.

    // ---------- VARIANTE B — appena si accende l'opt-in (authUid) ----------
    // Ordine di deploy OBBLIGATO: (1) `ensureSignedIn()` in tutti i client che scrivono
    // `partite` (hub + ruzzle + gameof15 + neonwar + i tre giochi nuovi — per i nuovi
    // basta `?auth=anon`), (2) deploy del codice, (3) deploy di queste regole.
    // Altrimenti i giochi legacy restano fuori e le scritture vengono rifiutate.
    /*

                    && request.resource.data.authUid == request.auth.uid
                    && statoGiusto() && durataSana();
      allow update: if request.auth != null
                     && (resource.data.authUid == request.auth.uid    // l'host della partita
                         || resource.data.authUid == null             // compatibilità: documenti

  }
}

// Cosa NON si ottiene con queste regole, così da non scambiarle per anti-cheat:
//   • i punteggi li calcola il client: `authUid` dimostra che la scrittura arriva da un
//     client autenticato, non che il punteggio sia onesto;
//   • finché `partecipanti` sono nomi, nessuna regola può verificare che chi scrive sia
//     davvero nella partita; quello richiede una migrazione (uid dentro `partecipanti`),
//     ed è una decisione da prendersi insieme perché tocca i documenti aperti;
//   • l'unica cosa che sposta l'autorità dal browser al server è una Cloud Function
//     (o un realtime server) che validi le mosse contro il dizionario.

// Indici: con le query attuali non serve alcun indice composito nuovo
//   • `partite` dove `partecipanti array-contains X` (hub, storico) → un filtro solo
//   • `partite` dove `gioco == Y && stato in [...]` (coda istantanea Bomba) → due eguaglianze,
//     servite dagli indici singoli (zigzag merge)
// Se le liste venissero ordinate in query (oggi si ordina in JavaScript apposta):
//   collection: partite   fields: gioco ASC, createdAt DESC
//   collection: partite   fields: partecipanti ARRAY, stato ASC, createdAt DESC
